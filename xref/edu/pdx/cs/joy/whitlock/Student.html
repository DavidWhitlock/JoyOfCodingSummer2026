9   /**
10   * This class represents a <code>Student</code>.
11   */
12  public class Student extends Human {
13  
14    private final double gpa;
15    private final ArrayList<String> classes;
16    private final Gender gender;
17  
18    /**
19     * Creates a new <code>Student</code>
20     *
21     * @param name
22     *        The student's name
23     * @param classes
24     *        The names of the classes the student is taking.  A student
25     *        may take zero or more classes.
26     * @param gpa
27     *        The student's grade point average
28     * @param gender
29     *        The student's gender ("male", "female", or "other", case-insensitive)
30     */
31    public Student(String name, ArrayList<String> classes, double gpa, String gender) {
32      super(name);
33      this.classes = classes;
34      this.gender = getGenderFor(gender);
35  
36      if (gpa < 0.0 || gpa > 4.0) {
37        throw new InvalidGPAException(gpa);
38      }
39  
40      this.gpa = gpa;
41    }
42  
43    private Gender getGenderFor(String gender) {
44      if (gender.equalsIgnoreCase("Female")) {
45        return Gender.FEMALE;
46  
47      } else if (gender.equalsIgnoreCase("Male")) {
48        return Gender.MALE;
49  
50      } else if (gender.equalsIgnoreCase("Other")) {
51        return Gender.OTHER;
52  
53      } else {
54        throw new UnsupportedGenderException(gender);
55      }
56    }
57  
58    /**
59     * All students say "This class is too much work"
60     */
61    @Override
62    public String says() {
63      return "This class is too much work";
64    }
65  
66    /**
67     * Returns a <code>String</code> that describes this
68     * <code>Student</code>.
69     */
70    public String toString() {
71      StringBuilder sb = new StringBuilder()
72        .append(this.getName())
73        .append(" has a GPA of ")
74        .append(this.gpa)
75        .append(" and is taking ")
76        .append(getClassesDescription()).append("  ")
77        .append(getPronoun()).append(" say");
78      if (this.gender == Gender.FEMALE || this.gender == Gender.MALE) {
79        sb.append("s");
80      }
81      sb.append(" \"");
82      sb.append(this.says());
83      sb.append("\".");
84      return sb.toString();
85    }
86  
87    private @NonNull String getClassesDescription() {
88      int numberOfClasses = this.classes.size();
89      StringBuilder sb = new StringBuilder().append(numberOfClasses);
90      sb.append(" class");
91      if (numberOfClasses == 0) {
92        sb.append("es");
93  
94      } else if (numberOfClasses == 1) {
95        sb.append(": ");
96        sb.append(classes.getFirst());
97  
98      } else if (numberOfClasses == 2) {
99        sb.append("es: ");
100       sb.append(classes.getFirst());
101       sb.append(" and ");
102       sb.append(classes.getLast());
103 
104     } else {
105       sb.append("es: ");
106 
107       for (int i = 0; i < numberOfClasses; i++) {
108         sb.append(classes.get(i));
109         if (i < numberOfClasses - 1) {
110           sb.append(", ");
111         }
112         if (i == numberOfClasses - 2) {
113           sb.append("and ");
114         }
115       }
116     }
117 
118     sb.append('.');
119     return sb.toString();
120   }

122   /**
123    * Main program that parses the command line, creates a
124    * <code>Student</code>, and prints a description of the student to
125    * standard out by invoking its <code>toString</code> method.
126    */
127   public static void main(String[] args) {
128     if (args.length == 0) {
129       System.err.println("Missing name");
130       return;
131     }
132 
133     if (args.length == 1) {
134       System.err.println("Missing gender");
135       return;
136     }
137 
138     if (args.length == 2) {
139       System.err.println("Missing GPA");
140       return;
141     }
142 
143     String name = args[0];
144     String gender = args[1];
145     double gpa;
146     try {
147       gpa = Double.parseDouble(args[2]);
148 
149     } catch (NumberFormatException e) {
150       System.err.println("Invalid GPA: " + args[2] + ".  GPA must be a number.");
151       return;
152     }
153 
154     ArrayList<String> classes = new ArrayList<>(Arrays.asList(args).subList(3, args.length));
155 
156     Student student;
157     try {
158       student = new Student(name, classes, gpa, gender);
159 
160     } catch (InvalidGPAException e) {
161       System.err.println("Invalid GPA: " + e.getInvalidGPA() + ".  Valid GPAS are in the range of 0.0-4.0");
162       return;
163 
164     } catch (UnsupportedGenderException e) {
165       System.err.println("Unsupported Gender: " + e.getUnsupportedGender());
166       return;
167     }
168     System.out.println(student);
169   }
170 
171   public String getPronoun() {
172     return switch (this.gender) {
173       case FEMALE -> "She";
174       case MALE -> "He";
175       case OTHER -> "They";
176     };
177   }
